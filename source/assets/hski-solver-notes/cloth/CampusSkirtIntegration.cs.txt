 private void ApplyCampusSkirt() {
  if(!UseNativeSkirt)return;

   campusSkirtBody=activeWardrobeBody;
   var bones=activeWardrobeBody.GetComponentsInChildren<Transform>(true).ToDictionary(x=>x.name,x=>x);
   var asset=Resources.Load<TextAsset>("WardrobeSkirts/"+CampusBodyBundle);
   var config=campusSkirtConfig=JsonUtility.FromJson<SkirtSettings>(asset.text);
   campusSkirtDrivers=config.drivers.Select(s=>new SkirtDriver{bone=bones[s.bone],reference=bones[s.reference],setting=s}).ToArray();
   if(!RuntimePerformance.Has("-skirtRootOnly")){

    var paddingArg=Environment.GetCommandLineArgs().FirstOrDefault(a=>a.StartsWith("-skirtPadding="));

     foreach(var collider in campusSkirtSwing.bodyColliders){collider.radius+=padding;if(collider.endRadius>=0)collider.endRadius+=padding;}

   }

  }

  if(campusSkirtSwing){

  }

 private int skirtAuditFrame;
 private bool skirtAuditDone;
 private int skirtAuditCycles=1;
 private float SkirtAuditTime=>skirtAuditCycles>1?Mathf.Repeat(skirtAuditFrame/60f,skirtAuditClip.length):Mathf.Min(skirtAuditFrame/60f,skirtAuditClip.length);
 private string skirtAuditOutput;

 private Mesh skirtAuditMesh;
 private SkinnedMeshRenderer skirtAuditRenderer;
 private readonly List<Vector3> skirtAuditVertices=new List<Vector3>();
 private readonly List<string> skirtAuditRows=new List<string>();
 private void InitializeSkirtAudit() {
  if(skirtAuditClip==null){
   Time.captureDeltaTime=1f/60f;QualitySettings.vSyncCount=0;Application.targetFrameRate=-1;
   skirtAuditClip=RuntimePerformance.Has("-skirtWalk")?WalkClip:RunClip;
   if(RuntimePerformance.Has("-skirtLong"))skirtAuditCycles=3;
   skirtAuditOutput=Environment.GetCommandLineArgs().FirstOrDefault(s=>s.StartsWith("-skirtOutput="))?.Substring(13)??Path.Combine(Application.persistentDataPath,"SkirtAudit");
   Directory.CreateDirectory(skirtAuditOutput);
   skirtAuditRenderer=activeWardrobeBody.GetComponentsInChildren<SkinnedMeshRenderer>().First(r=>r.bones.Any(b=>b&&b.name.Contains("Skirt")));
   skirtAuditMesh=new Mesh();skirtAuditMesh.MarkDynamic();
   skirtAuditRenderer.BakeMesh(skirtAuditMesh);
   var mesh=skirtAuditRenderer.sharedMesh;var weights=mesh.boneWeights;var bones=skirtAuditRenderer.bones;
   using(var w=new BinaryWriter(File.Create(Path.Combine(skirtAuditOutput,"topology.bin")))){
    w.Write(mesh.vertexCount);var tri=skirtAuditMesh.triangles;if(tri.Length==0)throw new InvalidOperationException("Baked skirt topology is empty");w.Write(tri.Length);foreach(int t in tri)w.Write(t);
    foreach(var bw in weights){float skirt=0,left=0,right=0;
     int[] ids={bw.boneIndex0,bw.boneIndex1,bw.boneIndex2,bw.boneIndex3};float[] ws={bw.weight0,bw.weight1,bw.weight2,bw.weight3};
     for(int j=0;j<4;j++){string n=bones[ids[j]]?bones[ids[j]].name:"";if(n.Contains("Skirt"))skirt+=ws[j];else if(n.Contains("Leg")){if(n.StartsWith("Left"))left+=ws[j];if(n.StartsWith("Right"))right+=ws[j];}}
     w.Write((byte)(skirt>.45f?1:left>.45f?2:right>.45f?3:0));
    }
   }
   if(skirtAuditRows.Count==0)skirtAuditRows.Add("frame,time,clip,mode,state");
  }
 }
 private void UpdateSkirtAudit() {
  if(skirtAuditDone)return;
  InitializeSkirtAudit();
  float time=SkirtAuditTime;
  SampleAndCorrect(skirtAuditClip,time,false);locomotionTime=time;walkTime=time;skirtPoseBlend=1;skirtPoseBlendVelocity=0;
  ApplyAuthoredSkirtPose(1,true,skirtAuditClip==WalkClip);
  wasGrounded=true;verticalVelocity=-2;stateName="Skirt validation";
 }
 private void RecordSkirtAudit() {
  if((!SkirtAuditActive&&!SkirtLiveAudit)||skirtAuditDone)return;
  if(SkirtLiveAudit){
   if(!UseNativeSkirt){skirtAuditFrame++;return;}
   if(!skirtAuditRenderer){skirtAuditClip=null;InitializeSkirtAudit();}
  }
  if(!skirtAuditRenderer)return;
  playerCamera.transform.position=playerRoot.transform.TransformPoint(new Vector3(.9f,1.0f,2.7f));
  playerCamera.transform.LookAt(playerRoot.transform.TransformPoint(new Vector3(0,.85f,0)));
  playerCamera.fieldOfView=32;
  skirtAuditRenderer.BakeMesh(skirtAuditMesh);skirtAuditMesh.GetVertices(skirtAuditVertices);
  using(var w=new BinaryWriter(File.Create(Path.Combine(skirtAuditOutput,$"frame_{skirtAuditFrame:000}.bin"))))foreach(var v in skirtAuditVertices){var p=playerRoot.transform.InverseTransformPoint(skirtAuditRenderer.transform.TransformPoint(v));w.Write(p.x);w.Write(p.y);w.Write(p.z);}
  skirtAuditRows.Add(FormattableString.Invariant($"{skirtAuditFrame},{(SkirtLiveAudit?selfTestTime:SkirtAuditTime):F6},{(SkirtLiveAudit?"self-test":skirtAuditClip.name)},{(LegacySkirt?"legacy":campusSkirtSwing?"native-root-and-swing":"native-root")},{stateName}"));
  int last=SkirtLiveAudit?839:Mathf.CeilToInt(skirtAuditClip.length*60*skirtAuditCycles);
  if(skirtAuditFrame==9||skirtAuditFrame==20||skirtAuditFrame==40||skirtAuditFrame==49||skirtAuditFrame==60||skirtAuditFrame>=last)CaptureSkirtFrame(skirtAuditFrame,skirtAuditFrame>=last);
  skirtAuditFrame++;
 }
 private void CaptureSkirtFrame(int frame,bool last){
  var target=RenderTexture.GetTemporary(1280,720,24,RenderTextureFormat.ARGB32);var oldTarget=playerCamera.targetTexture;var oldActive=RenderTexture.active;
  playerCamera.targetTexture=target;playerCamera.Render();RenderTexture.active=target;
  var image=new Texture2D(1280,720,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,1280,720),0,0);image.Apply();
  File.WriteAllBytes(Path.Combine(skirtAuditOutput,$"view_{frame:000}.png"),image.EncodeToPNG());Destroy(image);playerCamera.targetTexture=oldTarget;RenderTexture.active=oldActive;RenderTexture.ReleaseTemporary(target);
  if(last){skirtAuditDone=true;File.WriteAllLines(Path.Combine(skirtAuditOutput,"frames.csv"),skirtAuditRows);Debug.Log("SKIRT_AUDIT_DONE frames="+(skirtAuditRows.Count-1));if(!SkirtLiveAudit)Application.Quit();}}
