    private void Update()
    {
        if (!ready) return;
        if (reference0514) return;
        if (SkirtAuditActive) { UpdateSkirtAudit(); return; }
        UpdateWardrobeSkirtLifecycle();
        UpdateCampusHairControls();

        if (!selfTestActive && Input.GetMouseButtonDown(1)) Cursor.lockState = CursorLockMode.Locked;
        if (!selfTestActive && Input.GetMouseButtonUp(1)) Cursor.lockState = CursorLockMode.None;
        if (!selfTestActive && Input.GetMouseButton(1))
        {
            yaw += Input.GetAxis("Mouse X") * 150f * Time.deltaTime;
            pitch = Mathf.Clamp(pitch - Input.GetAxis("Mouse Y") * 110f * Time.deltaTime, -10f, 55f);
        }
        if (!selfTestActive)
            cameraDistance = Mathf.Clamp(cameraDistance - Input.mouseScrollDelta.y * 0.35f, 2.4f, 7f);
        if (!selfTestActive) HandleWardrobeInput();

        if (selfTestActive) selfTestTime += Time.deltaTime;
        if (selfTestActive && RuntimePerformance.Has("-roomOrbitSelfTest"))
        {
            yaw = 180f + Mathf.Floor(selfTestTime) * 45f;
            pitch = 8f;
            cameraDistance = 2.8f;
        }
        if (selfTestActive && wardrobeSelfTestActive) UpdateWardrobeSelfTest();
        var input = selfTestActive
            ? (wardrobeSelfTestActive ? Vector2.zero : SelfTestInput(selfTestTime))
            : Vector2.ClampMagnitude(new Vector2(Input.GetAxisRaw("Horizontal"), Input.GetAxisRaw("Vertical")), 1f);
        if (selfTestActive && RuntimePerformance.Has("-roomOrbitSelfTest")) input = Vector2.zero;
        if (RuntimePerformance.Has("-renderPerformanceTest")) input = Vector2.zero;
        var cameraYaw = Quaternion.Euler(0f, yaw, 0f);
        var direction = cameraYaw * new Vector3(input.x, 0f, input.y);
        var grounded = ProbeGrounded();
        if (grounded && verticalVelocity < 0f) verticalVelocity = -2f;
        var jumpPressed = selfTestActive
            ? !wardrobeSelfTestActive && selfTestTime >= 8.0f && selfTestTime - Time.deltaTime < 8.0f
            : Input.GetKeyDown(KeyCode.Space);
        if (grounded && jumpPressed && !RuntimePerformance.Has("-renderPerformanceTest"))
        {
            verticalVelocity = Mathf.Sqrt(JumpHeight * -2f * Gravity);
            grounded = false;
            airTime = 0f;
        }
        verticalVelocity += Gravity * Time.deltaTime;

        var walk = selfTestActive
            ? !wardrobeSelfTestActive && selfTestTime >= 1f && selfTestTime < 5f
            : Input.GetKey(KeyCode.LeftControl) || Input.GetKey(KeyCode.RightControl);
        var sprint = selfTestActive
            ? !wardrobeSelfTestActive && selfTestTime >= 7f && selfTestTime < 8f
            : Input.GetKey(KeyCode.LeftShift) || Input.GetKey(KeyCode.RightShift);
        var moveSpeed = walk ? WalkSpeed : sprint ? SprintSpeed : RunSpeed;
        var velocity = direction * moveSpeed;
        velocity.y = verticalVelocity;
        characterController.Move(velocity * Time.deltaTime);

        if (direction.sqrMagnitude > 0.001f)
        {
            var desiredRotation = Quaternion.LookRotation(direction, Vector3.up);
            playerRoot.transform.rotation = Quaternion.RotateTowards(
                playerRoot.transform.rotation, desiredRotation, TurnSpeed * Time.deltaTime);
        }

        grounded = ProbeGrounded();
        if (grounded && verticalVelocity <= 0f)
        {
            lastSafeGroundPosition = playerRoot.transform.position;
            hasSafeGroundPosition = true;
        }
        else if (hasSafeGroundPosition &&
                 playerRoot.transform.position.y < lastSafeGroundPosition.y - 4f)
        {
            characterController.enabled = false;
            playerRoot.transform.position = lastSafeGroundPosition + Vector3.up * 0.08f;
            characterController.enabled = true;
            verticalVelocity = -2f;
            groundedGrace = 0.08f;
            grounded = true;
            Debug.LogWarning($"LOCOMOTION_GROUND_RECOVERY position={lastSafeGroundPosition}");
        }
        if (!grounded)
        {
            airTime += Time.deltaTime;
            SampleAndCorrect(JumpClip, Mathf.Min(airTime, JumpClip.length), false);
            stateName = verticalVelocity >= 0f ? "Jump" : "Fall (hold Jump end)";
        }
        else if (input.sqrMagnitude > 0.001f)
        {
            if (walk)
            {
                walkTime += Time.deltaTime;
                SampleAndCorrect(WalkClip, Mathf.Repeat(walkTime, WalkClip.length), false);
                // Preserve the native Gakumas walk, including knee and ankle articulation.
                stateName = "Walk " + ResolveFourWayName(input);
            }
            else
            {
                var playbackRate = sprint ? 1.35f : 1f;
                locomotionTime += Time.deltaTime * playbackRate;
                // Native Gakumas run: preserve authored pelvis, arms, knees and ankles.
                SampleAndCorrect(RunClip, Mathf.Repeat(locomotionTime, RunClip.length), false);
                stateName = sprint ? "Sprint (Run 1.35x)" : "Run";
            }
        }
        else
        {
            idleTime += Time.deltaTime;
            SampleAndCorrect(IdleClip, Mathf.Repeat(idleTime, IdleClip.length), false);
            stateName = wasGrounded ? "Idle" : "Land → Idle";
        }
        long skirtTick=RuntimePerformance.Tick(), skirtBytes=RuntimePerformance.Bytes();
        ApplyAuthoredSkirtPose(input.magnitude, grounded, walk);
        RuntimePerformance.Record("SkirtCorrection",skirtTick,skirtBytes);
        if (selfTestActive && !RuntimePerformance.Enabled && grounded && input.sqrMagnitude > 0.001f && !walk)
            UpdateSkirtMetrics(input.magnitude);
        wasGrounded = grounded;
        if (selfTestActive && !CampusHairBenchmark) UpdateSelfTestEvidence();
    }

    private void LateUpdate()
    {
        if (!ready) return;
        long lateTick=RuntimePerformance.Tick(), lateBytes=RuntimePerformance.Bytes();
        // Ordered pose pipeline: sampled base pose -> authored skirt -> clothing sync
        // -> world-space foot clearance -> clothing resync -> shader matrices -> camera.
        // The legacy branch retains the original competing global driver for A/B tests.
        if(RuntimePerformance.Legacy)
        {
            Shader.SetGlobalVector("_MatCapRimLight", Vector4.zero);
            Shader.SetGlobalColor("_MatCapRimColor", Color.clear);
        }
        SyncWardrobeBones();
        long soleTick=RuntimePerformance.Tick(), soleBytes=RuntimePerformance.Bytes();
        AlignVisualFeet();
        RuntimePerformance.Record("Sole",soleTick,soleBytes);
        SyncWardrobeBones();
        ApplyCampusSkirt();
        StepCampusHair();
        // Manual animation sampling has now updated the head; override the hidden
        // comparison actor's global driver before the frame is rendered.
        long globalsTick=RuntimePerformance.Tick(), globalsBytes=RuntimePerformance.Bytes();
        if (characterAttributes != null) characterAttributes.ApplyNow();
        RuntimePerformance.Record("CharacterGlobals",globalsTick,globalsBytes);
        long cameraTick=RuntimePerformance.Tick(), cameraBytes=RuntimePerformance.Bytes();
        UpdateCamera(false);
        RuntimePerformance.Record("Camera",cameraTick,cameraBytes);
        RuntimePerformance.Record("LateUpdate",lateTick,lateBytes);
        RecordSkirtAudit();
    }

    private void SampleAndCorrect(AnimationClip clip, float time, bool runCorrection)
    {
        long tick=RuntimePerformance.Tick(), bytes=RuntimePerformance.Bytes();
        RestoreCampusHairPose();
        RestoreCampusSkirtPose();
        clip.SampleAnimation(visual, time);
        RuntimePerformance.Record("AnimationSample",tick,bytes);
        visual.transform.SetPositionAndRotation(playerRoot.transform.position, playerRoot.transform.rotation);
        ResetSkirtPose();
        if (!runCorrection) return;

        var right = playerRoot.transform.right;
        RotateWorld(rig.Hips, right, 3f);
        RotateWorld(rig.Chest, right, 6.5f);
        var phase = time / Mathf.Max(0.001f, RunClip.length) * Mathf.PI * 2f;
        var swing = Mathf.Sin(phase) * 11f;
        RotateWorld(rig.LeftUpperArm, right, swing);
        RotateWorld(rig.RightUpperArm, right, -swing);
        RotateWorld(rig.LeftLowerArm, right, swing * 0.28f + 3f);
        RotateWorld(rig.RightLowerArm, right, -swing * 0.28f + 3f);
    }

    private void ResetSkirtPose()
    {
        if (skirtChains == null) return;
        foreach (var chain in skirtChains)
            for (var i = 0; i < chain.Bones.Length; i++)
                chain.Bones[i].localRotation = chain.RestLocalRotations[i];
        foreach (var chain in skirtChains)
            for (var i = 0; i < chain.Bones.Length; i++)
                chain.Bones[i].localPosition = chain.RestLocalPositions[i];
    }

    private void ApplyAuthoredSkirtPose(float movementAmount, bool grounded, bool walking)
    {
        if (UseNativeSkirt) return;
        if (skirtChains == null || skirtChains.Length == 0) return;

        // Authored run-cycle skirt keys. Only the panel beside the forward leg opens;
        // the opposite panel returns toward rest. SmoothDamp handles state transitions,
        // and smooth interpolation between keys avoids the previous radial "explosion".
        var targetBlend = grounded ? movementAmount * (walking ? 0.55f : 1f) : 0.38f;
        skirtPoseBlend = Mathf.SmoothDamp(skirtPoseBlend, targetBlend,
            ref skirtPoseBlendVelocity, grounded ? 0.08f : 0.16f);
        var phase = Mathf.Repeat(locomotionTime / Mathf.Max(0.001f, RunClip.length), 1f);

        foreach (var chain in skirtChains)
        {
            var panelPhase = chain.Right ? Mathf.Repeat(phase + 0.5f, 1f) : phase;
            var sagittal = chain.Front
                ? EvaluateLoopKeys(SkirtFrontKeys, panelPhase)
                : chain.Back ? EvaluateLoopKeys(SkirtBackKeys, panelPhase) : 0f;
            var frontClearance = 0f;
            if (chain.Front)
            {
                var upperLeg = chain.Left ? rig.LeftUpperLeg : rig.RightUpperLeg;
                var lowerLeg = chain.Left ? rig.LeftLowerLeg : rig.RightLowerLeg;
                var thighForward = Vector3.Dot((lowerLeg.position - upperLeg.position).normalized,
                    playerRoot.transform.forward);
                var forwardAmount = Mathf.InverseLerp(-0.12f, 0.42f, thighForward);
                sagittal += forwardAmount * (walking ? 0.8f : 2.0f);
                if (!walking)
                    frontClearance = (0.015f + forwardAmount * 0.020f) * skirtPoseBlend;
            }
            var lateralMagnitude = chain.Side ? EvaluateLoopKeys(SkirtLateralKeys, panelPhase) : 0f;
            var lateral = chain.Left ? -lateralMagnitude : chain.Right ? lateralMagnitude : 0f;
            // Keep the two central front panels meeting at the hem. Retargeted
            // forward thigh motion otherwise separates them and exposes a tall
            // triangular gap even when neither panel actually intersects the leg.
            if (chain.Front && !chain.Side)
                lateral += chain.Left ? 4f : -4f;
            sagittal *= skirtPoseBlend;
            lateral *= skirtPoseBlend;

            if (chain.Bones.Length > 0)
            {
                RotateWorld(chain.Bones[0], playerRoot.transform.right, sagittal);
                RotateWorld(chain.Bones[0], playerRoot.transform.forward, lateral);
                if (frontClearance > 0f)
                    chain.Bones[0].position += playerRoot.transform.forward * frontClearance;
            }
            // A very small follow-through on the first child keeps the hem from
            // looking rigid without accumulating rotations down the entire chain.
            if (chain.Bones.Length > 1)
            {
                RotateWorld(chain.Bones[1], playerRoot.transform.right, sagittal * 0.08f);
                RotateWorld(chain.Bones[1], playerRoot.transform.forward, lateral * 0.08f);
            }
        }
    }

    private void ApplyLegStrideCorrection(bool walking)
    {
        if (!legReferencePoseReady) return;

        if (walking)
        {
            // Leg edits must not rotate the sampled ankle a second time through
            // the parent chain. Keep the source heel/toe roll in world space.
            var leftFootRotation = rig.LeftFoot.rotation;
            var rightFootRotation = rig.RightFoot.rotation;
            // Re-author the lower-body walk on HSKI's idle reference pose. The
            // retargeted source can put both thighs in front of the pelvis because
            // its hip translation does not match HSKI's shorter proportions.
            // Strictly opposite thigh phases plus rear-leg knee flexion produce a
            // compact walk while the source clip still supplies torso/arm motion.
            var phase = Mathf.Repeat(walkTime / Mathf.Max(0.001f, WalkClip.length), 1f);
            var swing = Mathf.Sin(phase * Mathf.PI * 2f);
            rig.LeftUpperLeg.localRotation = leftUpperLegIdleLocalRotation;
            rig.RightUpperLeg.localRotation = rightUpperLegIdleLocalRotation;
            rig.LeftLowerLeg.localRotation = leftLowerLegIdleLocalRotation;
            rig.RightLowerLeg.localRotation = rightLowerLegIdleLocalRotation;

            RotateWorld(rig.LeftUpperLeg, playerRoot.transform.right, -swing * 15f);
            RotateWorld(rig.RightUpperLeg, playerRoot.transform.right, swing * 15f);
            RotateWorld(rig.LeftLowerLeg, playerRoot.transform.right,
                3f + Mathf.Max(0f, -swing) * 15f);
            RotateWorld(rig.RightLowerLeg, playerRoot.transform.right,
                3f + Mathf.Max(0f, swing) * 15f);
            rig.LeftFoot.rotation = leftFootRotation;
            rig.RightFoot.rotation = rightFootRotation;
            return;
        }

        // The run source only needs a mild forward-swing compression.
        CompressForwardThigh(rig.LeftUpperLeg, rig.LeftLowerLeg,
            leftUpperLegIdleLocalRotation, 0.30f, 0.40f);
        CompressForwardThigh(rig.RightUpperLeg, rig.RightLowerLeg,
            rightUpperLegIdleLocalRotation, 0.30f, 0.40f);
    }

    private void CompressForwardThigh(Transform upperLeg, Transform lowerLeg,
        Quaternion idleLocalRotation, float maximumBlend, float correctionStart)
    {
        var thighForward = Vector3.Dot((lowerLeg.position - upperLeg.position).normalized,
            playerRoot.transform.forward);
        var excess = Mathf.InverseLerp(correctionStart, 0.82f, thighForward);
        upperLeg.localRotation = Quaternion.Slerp(upperLeg.localRotation,
            idleLocalRotation, excess * maximumBlend);
    }

    private static float EvaluateLoopKeys(float[] keys, float normalizedTime)
    {
        var position = Mathf.Repeat(normalizedTime, 1f) * (keys.Length - 1);
        var index = Mathf.Min(Mathf.FloorToInt(position), keys.Length - 2);
        var t = Mathf.SmoothStep(0f, 1f, position - index);
        return Mathf.Lerp(keys[index], keys[index + 1], t);
    }

    private void UpdateSkirtMetrics(float movementAmount)
    {
        var sampleIndex = Mathf.FloorToInt((selfTestTime - 5.15f) / 0.14f);
        if (sampleIndex < 0 || sampleIndex > 5) return;
        var key = sampleIndex.ToString();
        if (!skirtMetricSamples.Add(key)) return;

        ResetSkirtPose();
        var baseline = CountSkirtLegPenetrationCandidates();
        ApplyAuthoredSkirtPose(movementAmount, true, false);
        var corrected = CountSkirtLegPenetrationCandidates();
        var reduction = baseline > 0 ? (baseline - corrected) * 100f / baseline : 0f;
        var line = $"phase_{sampleIndex},{selfTestTime:F3},{baseline},{corrected},{reduction:F1}";
        skirtMetricTrace.Add(line);
        Debug.Log("LOCOMOTION_SKIRT_METRIC " + line);
    }

    private int CountSkirtLegPenetrationCandidates()
    {
        var count = 0;
        foreach (var renderer in visual.GetComponentsInChildren<SkinnedMeshRenderer>(true))
        {
            var source = renderer.sharedMesh;
            if (source == null || source.boneWeights.Length != source.vertexCount) continue;
            var boneIsSkirt = renderer.bones.Select(bone => bone != null &&
                bone.name.IndexOf("Skirt", StringComparison.OrdinalIgnoreCase) >= 0).ToArray();
            if (!boneIsSkirt.Any(value => value)) continue;

            var baked = new Mesh();
            renderer.BakeMesh(baked);
            var vertices = baked.vertices;
            var weights = source.boneWeights;
            for (var i = 0; i < vertices.Length; i++)
            {
                var weight = weights[i];
                var skirtWeight = 0f;
                if (weight.boneIndex0 < boneIsSkirt.Length && boneIsSkirt[weight.boneIndex0]) skirtWeight += weight.weight0;
                if (weight.boneIndex1 < boneIsSkirt.Length && boneIsSkirt[weight.boneIndex1]) skirtWeight += weight.weight1;
                if (weight.boneIndex2 < boneIsSkirt.Length && boneIsSkirt[weight.boneIndex2]) skirtWeight += weight.weight2;
                if (weight.boneIndex3 < boneIsSkirt.Length && boneIsSkirt[weight.boneIndex3]) skirtWeight += weight.weight3;
                if (skirtWeight < 0.45f) continue;

                var world = renderer.transform.TransformPoint(vertices[i]);
                var hipLimit = Mathf.Min(rig.LeftUpperLeg.position.y, rig.RightUpperLeg.position.y) - 0.09f;
                if (world.y > hipLimit) continue;
                if (DistanceToSegment(world, rig.LeftUpperLeg.position, rig.LeftLowerLeg.position) < 0.105f
                    || DistanceToSegment(world, rig.RightUpperLeg.position, rig.RightLowerLeg.position) < 0.105f)
                    count++;
            }
            Destroy(baked);
        }
        return count;
    }

    private static float DistanceToSegment(Vector3 point, Vector3 start, Vector3 end)
    {
        var segment = end - start;
        var t = Mathf.Clamp01(Vector3.Dot(point - start, segment) / Mathf.Max(0.000001f, segment.sqrMagnitude));
        return Vector3.Distance(point, start + segment * t);
    }

    private static SkirtChain[] BuildSkirtChains(Transform root)
    {
        var all = root.GetComponentsInChildren<Transform>(true);
        var anchors = all.Where(transform =>
        {
            var name = transform.name;
            if (name.IndexOf("Skirt", StringComparison.OrdinalIgnoreCase) < 0) return false;
            return name.EndsWith("_A", StringComparison.OrdinalIgnoreCase);
        }).ToArray();

        return anchors.Select(anchor =>
        {
            var name = anchor.name;
            var bones = anchor.GetComponentsInChildren<Transform>(true)
                .Where(transform => transform.name.IndexOf("Skirt", StringComparison.OrdinalIgnoreCase) >= 0)
                .Take(5).ToArray();
            return new SkirtChain
            {
                Name = name,
                Bones = bones,
                RestLocalPositions = bones.Select(bone => bone.localPosition).ToArray(),
                RestLocalRotations = bones.Select(bone => bone.localRotation).ToArray(),
                Left = name.IndexOf("Left", StringComparison.OrdinalIgnoreCase) >= 0,
                Right = name.IndexOf("Right", StringComparison.OrdinalIgnoreCase) >= 0,
                Front = name.IndexOf("Front", StringComparison.OrdinalIgnoreCase) >= 0,
                Back = name.IndexOf("Back", StringComparison.OrdinalIgnoreCase) >= 0,
                Side = name.IndexOf("Side", StringComparison.OrdinalIgnoreCase) >= 0
            };
        }).ToArray();
    }

    private void AlignVisualFeet()
    {
        if (rig.LeftFoot == null || rig.RightFoot == null) return;
        bool grounded = !ready || (wasGrounded && verticalVelocity <= 0f);
        if (!grounded)
        {
            // Preserve the sampled jump trajectory; no ground projection in flight.
            soleLift = 0f;
            if (selfTestActive && !RuntimePerformance.Enabled) soleTrace.Add($"{selfTestTime:F4},{stateName},false,0,0,0,0,0");
            return;
        }
        var body = activeWardrobeBody != null ? activeWardrobeBody : visual;
        if (soleRenderers == null || soleBody != body)
        {
            soleBody = body;
            soleRenderers = body.GetComponentsInChildren<SkinnedMeshRenderer>(true);
        }
        if (soleMesh == null) { soleMesh = new Mesh { name = "Runtime sole probe" }; soleMesh.MarkDynamic(); }
        RaycastHit leftGround, rightGround;
        bool leftValid = FindSoleGround(rig.LeftFoot.position, out leftGround);
        bool rightValid = FindSoleGround(rig.RightFoot.position, out rightGround);
        RaycastHit support;
        float supportOffset = FindSoleGround(playerRoot.transform.position, out support)
            ? support.point.y - playerRoot.transform.position.y : 0f;
        float minimum = float.PositiveInfinity;
        float lift = 0f;
        int samples = 0;
        foreach (var renderer in soleRenderers)
        {
            if (renderer == null || !renderer.enabled || !renderer.gameObject.activeInHierarchy || renderer.sharedMesh == null) continue;
            // Bake into our own readable mesh: never read vertices from runtime AB meshes.
            string bundle = activeWardrobeBody != null && bodyOptions.Length>0 ? bodyOptions[bodyOptionIndex].BundleName : null;
            bool sparse = !RuntimePerformance.Legacy && sparseSoles.Fill(renderer,bundle,sparseVertices);
            var vertices=sparse ? sparseVertices : soleVertices;
            if(!sparse){renderer.BakeMesh(soleMesh, false);soleMesh.GetVertices(soleVertices);}
            float rendererMinimum=float.PositiveInfinity, rendererLift=0f;int rendererSamples=0;
            EvaluateSoleVertices(renderer,vertices,sparse,supportOffset,leftGround,leftValid,rightGround,rightValid,
                ref rendererMinimum,ref rendererLift,ref rendererSamples);
            if(sparse && RuntimePerformance.Has("-validateSoles"))
            {
                renderer.BakeMesh(soleMesh,false);soleMesh.GetVertices(soleVertices);
                float exactMinimum=float.PositiveInfinity,exactLift=0f;int exactSamples=0;
                EvaluateSoleVertices(renderer,soleVertices,false,supportOffset,leftGround,leftValid,rightGround,rightValid,
                    ref exactMinimum,ref exactLift,ref exactSamples);
                float difference=Mathf.Abs(rendererLift-exactLift);
                maximumSoleDifference=Mathf.Max(maximumSoleDifference,difference);soleValidationFrames++;
                if(difference>0.002f)
                {
                    Debug.LogError($"SPARSE_SOLE_MISMATCH mesh={renderer.name} difference={difference:F6}; reverting renderer to BakeMesh");
                    sparseSoles.Reject(renderer);rendererMinimum=exactMinimum;rendererLift=exactLift;rendererSamples=exactSamples;
                }
            }
            minimum=Mathf.Min(minimum,rendererMinimum);lift=Mathf.Max(lift,rendererLift);samples+=rendererSamples;
        }
        soleLift = Mathf.Max(lift, Mathf.MoveTowards(soleLift, lift, Time.deltaTime * 1.2f));
        if (samples == 0) soleLift = 0f;
        float legacy = samples > 0 ? minimum - supportOffset + playerRoot.transform.position.y -
            Mathf.Min(rig.LeftFoot.position.y, rig.RightFoot.position.y) : 0f;
        visual.transform.position += Vector3.up * (supportOffset + soleLift);
        if (selfTestActive && samples > 0 && !RuntimePerformance.Enabled)
            soleTrace.Add($"{selfTestTime:F4},{stateName},true,{samples},{minimum:F5},{minimum + soleLift * Mathf.Min(leftValid ? leftGround.normal.y : 1f, rightValid ? rightGround.normal.y : 1f):F5},{legacy:F5},{soleLift:F5}");
    }
    private void EvaluateSoleVertices(SkinnedMeshRenderer renderer,List<Vector3> vertices,bool worldSpace,float supportOffset,
        RaycastHit leftGround,bool leftValid,RaycastHit rightGround,bool rightValid,ref float minimum,ref float lift,ref int samples)
    {
            // Cache native Transform reads outside the candidate loop.
            Vector3 leftPosition=rig.LeftFoot.position, rightPosition=rig.RightFoot.position;
            foreach (var vertex in vertices)
            {
                var world = (worldSpace ? vertex : renderer.transform.TransformPoint(vertex)) + Vector3.up * supportOffset;
                bool left = RuntimePerformance.Legacy
                    ? (world - rig.LeftFoot.position).sqrMagnitude < (world - rig.RightFoot.position).sqrMagnitude
                    : (world-leftPosition).sqrMagnitude < (world-rightPosition).sqrMagnitude;
                var foot = RuntimePerformance.Legacy ? (left ? rig.LeftFoot.position : rig.RightFoot.position) : (left ? leftPosition : rightPosition);
                if (world.y > foot.y + 0.12f || (world-foot).sqrMagnitude > 0.16f) continue;
                if (left ? !leftValid : !rightValid) continue;
                var hit = left ? leftGround : rightGround;
                float clearance = Vector3.Dot(world - hit.point, hit.normal);
                minimum = Mathf.Min(minimum, clearance);
                lift = Mathf.Max(lift, (0.003f-clearance)/Mathf.Max(0.3f, hit.normal.y));
                samples++;
            }
    }

    private bool FindSoleGround(Vector3 foot, out RaycastHit result)
    {
        // Only walkable surfaces near the controller's support height count.
        var origin = new Vector3(foot.x, playerRoot.transform.position.y + 0.45f, foot.z);
        int count = Physics.RaycastNonAlloc(origin, Vector3.down, soleHits, 0.95f, ~0, QueryTriggerInteraction.Ignore);
        result = default;
        float closest = float.PositiveInfinity;
        for (int i = 0; i < count; i++)
        {
            var hit = soleHits[i];
            if (hit.normal.y < Mathf.Cos(characterController.slopeLimit * Mathf.Deg2Rad) ||
                hit.transform.IsChildOf(playerRoot.transform) || hit.transform.IsChildOf(visual.transform) ||
                (activeWardrobeBody != null && hit.transform.IsChildOf(activeWardrobeBody.transform))) continue;
            if (hit.distance < closest) { closest = hit.distance; result = hit; }
        }
        return closest < float.PositiveInfinity;
    }

    }

    private WardrobeOption[] LoadWardrobeOptions(string root,
        (string label, string bundle, string owner, string iconBundle)[] entries, bool includeOriginal)
    {
        var options = new List<WardrobeOption>();
        if (includeOriginal)
            options.Add(new WardrobeOption { Label = "Original HSKI", BundleName = string.Empty, Owner = "HSKI", Prefab = null,
                Thumbnail = LoadWardrobeIcon(root, entries.FirstOrDefault(entry => entry.owner == "HSKI").iconBundle) });
        foreach (var entry in entries)
        {
            GameObject prefab = null;
            Mesh mesh = null;
            if (!string.IsNullOrEmpty(entry.bundle))
            {
                var bundle = LoadWardrobeBundle(Path.Combine(root, entry.bundle), true);
                prefab = bundle == null ? null : bundle.LoadAllAssets<GameObject>().FirstOrDefault();
                mesh = bundle == null ? null : bundle.LoadAllAssets<Mesh>().FirstOrDefault();
                if (prefab == null)
                {
                    Debug.LogWarning($"WARDROBE_OPTION_MISSING bundle={entry.bundle}");
                    continue;
                }
            }
            if (string.IsNullOrEmpty(entry.bundle) && includeOriginal) continue;
            options.Add(new WardrobeOption { Label = entry.label, BundleName = entry.bundle, Owner = entry.owner,
                Prefab = prefab, Mesh = mesh, Thumbnail = LoadWardrobeIcon(root, entry.iconBundle) });
        }
        return options.ToArray();
    }

    {

        return bundle == null ? null : bundle.LoadAllAssets<Texture2D>().FirstOrDefault();
    }

    private Texture2D RenderOutfitThumbnail(WardrobeOption option)
    {
        // Render the actual outfit when the dump contains no matching UI sprite.
        var instance = Instantiate(option.Prefab);
        instance.transform.position = new Vector3(10000f, -10000f, 0f);
        SetLayerRecursively(instance.transform, 29);
        var previewMaterials = new List<Material>();
        var cameraObject = new GameObject("Outfit Thumbnail Camera");
        var camera = cameraObject.AddComponent<Camera>();
        camera.enabled = false;
        camera.cullingMask = 1 << 29;
        camera.clearFlags = CameraClearFlags.SolidColor;
        camera.backgroundColor = Color.clear;
        camera.orthographic = true;
        camera.nearClipPlane = 0.01f;
        camera.farClipPlane = 20f;
        var renderers = instance.GetComponentsInChildren<Renderer>(true);
        var previewShader = Shader.Find("Universal Render Pipeline/Unlit");
        var target = RenderTexture.GetTemporary(256, 256, 24, RenderTextureFormat.ARGB32);
        var previous = RenderTexture.active;
        try
        {
            if (previewShader == null || renderers.Length == 0) return null;
            foreach (var renderer in renderers)
            {
                renderer.sharedMaterials = renderer.sharedMaterials.Select(source =>
                {
                    var material = new Material(previewShader);
                    if (source != null) material.SetTexture("_BaseMap", source.GetTexture("_BaseMap"));
                    material.SetColor("_BaseColor", Color.white);
                    material.SetFloat("_Cull", 0f);
                    previewMaterials.Add(material);
                    return material;
                }).ToArray();
                if (renderer is SkinnedMeshRenderer skinned) skinned.updateWhenOffscreen = true;
            }
            var bounds = renderers[0].bounds;
            foreach (var renderer in renderers.Skip(1)) bounds.Encapsulate(renderer.bounds);
            camera.orthographicSize = Mathf.Max(bounds.extents.y, bounds.extents.x) * 1.12f;
            camera.transform.position = bounds.center + Vector3.forward * 5f;
            camera.transform.LookAt(bounds.center);
            camera.targetTexture = target;
            camera.Render();
            RenderTexture.active = target;
            var thumbnail = new Texture2D(256, 256, TextureFormat.RGBA32, false) { name = option.BundleName + "_Preview" };
            RuntimeAssetOwner.For(gameObject).Own(thumbnail);
            thumbnail.ReadPixels(new Rect(0, 0, 256, 256), 0, 0);
            thumbnail.Apply();
            return thumbnail;
        }
        finally
        {
            camera.targetTexture = null;
            RenderTexture.active = previous;
            RenderTexture.ReleaseTemporary(target);
            instance.SetActive(false);
            Destroy(instance);
            Destroy(cameraObject);
            foreach (var material in previewMaterials) Destroy(material);
        }
    }

    private AssetBundle LoadWardrobeBundle(string path, bool reportMissing)
    {
        if (!File.Exists(path))
        {
            if (reportMissing) Debug.LogWarning($"WARDROBE_FILE_MISSING path={path}");
            return null;
        }
        var bundle = wardrobeCache.Load(path);
        if (bundle != null) wardrobeBundles.Add(bundle);
        else if (reportMissing) Debug.LogWarning($"WARDROBE_LOAD_FAILED path={path}");
        return bundle;
    }

    private void HandleWardrobeInput()
    {
        if (Input.GetKeyDown(KeyCode.B)) wardrobeOpen = !wardrobeOpen;
        if (Input.GetKeyDown(KeyCode.Q)) SetCharacterPreset(characterPresetIndex - 1);
        if (Input.GetKeyDown(KeyCode.E)) SetCharacterPreset(characterPresetIndex + 1);
        if (Input.GetKeyDown(KeyCode.Z)) SetBodyOption(bodyOptionIndex - 1);
        if (Input.GetKeyDown(KeyCode.X)) SetBodyOption(bodyOptionIndex + 1);
    }

    private void SetCharacterPreset(int index)
    {
        if (characterPresets.Length == 0) return;
        characterPresetIndex = WrapIndex(index, characterPresets.Length);
        var preset = characterPresets[characterPresetIndex];
        SetFaceOption(FindOption(faceOptions, preset.FaceBundle));
        SetBodyOption(FindOption(bodyOptions, preset.BodyBundle));
        SetHairOption(FindOption(hairOptions, preset.HairBundle));
        Debug.Log($"WARDROBE_CHARACTER_CHANGED index={characterPresetIndex} owner={preset.Owner}");
    }

    private void SetFaceOption(int index)
    {
        if (faceOptions.Length == 0) return;
        ReleaseCampusHair();
        ReleaseCampusSkirt();
        faceOptionIndex = WrapIndex(index, faceOptions.Length);
        if (activeWardrobeFace != null) activeWardrobeFace.SetActive(false);
        var option = faceOptions[faceOptionIndex];
        activeWardrobeFace = null;
        if (option.Prefab != null)
        {
            if (!faceInstanceCache.TryGetValue(option.BundleName, out activeWardrobeFace) || activeWardrobeFace == null)
            {
                activeWardrobeFace = CreateWardrobeInstance(option, "Face");
                faceInstanceCache[option.BundleName] = activeWardrobeFace;
            }
            else activeWardrobeFace.SetActive(true);
        }
        foreach (var renderer in originalFaceRenderers) renderer.enabled = activeWardrobeFace == null;
        RebuildWardrobeBoneBindings();
        Debug.Log($"WARDROBE_FACE_CHANGED index={faceOptionIndex} label={faceOptions[faceOptionIndex].Label}");
    }

    private void SetBodyOption(int index)
    {
        if (bodyOptions.Length == 0) return;
        ReleaseCampusHair();
        ReleaseCampusSkirt();
        bodyOptionIndex = WrapIndex(index, bodyOptions.Length);
        if (activeWardrobeBody != null) Destroy(activeWardrobeBody);
        activeWardrobeBody = CreateWardrobeInstance(bodyOptions[bodyOptionIndex], "Body");
        foreach (var renderer in originalBodyRenderers) renderer.enabled = RuntimePerformance.Legacy ? bodyOptionIndex == 0 : activeWardrobeBody == null;
        RebuildWardrobeBoneBindings();
        Debug.Log($"WARDROBE_BODY_CHANGED index={bodyOptionIndex} label={bodyOptions[bodyOptionIndex].Label}");
    }

    private void SetHairOption(int index)
    {
        if (hairOptions.Length == 0) return;

        ReleaseCampusSkirt();

    }

    private GameObject CreateWardrobeInstance(WardrobeOption option, string category)
    {
        if (option == null || option.Prefab == null) return null;
        var attachToHead = string.Equals(category, "Face", StringComparison.OrdinalIgnoreCase) ||
                           string.Equals(category, "Hair", StringComparison.OrdinalIgnoreCase);
        var parent = visual.transform;
        if (attachToHead && wardrobeTargetBones.TryGetValue("Head", out var head)) parent = head;
        var instance = Instantiate(option.Prefab, parent);
        instance.name = $"Wardrobe {option.Owner} {option.Label} {category}";
        if (string.Equals(category, "Face", StringComparison.OrdinalIgnoreCase))
        {
            // Face vertices use the source idol's body space; hair is Head-local.
            // Read the original prefab, independent of the currently worn outfit.
            var preset = characterPresets.First(item => item.Owner == option.Owner);
            var sourceBody = bodyOptions.First(item => item.BundleName == preset.BodyBundle).Prefab;
            var sourceHead = sourceBody.GetComponentsInChildren<Transform>(true).First(item => item.name == "Head");
            var sourcePosition = sourceBody.transform.InverseTransformPoint(sourceHead.position);
            var sourceRotation = Quaternion.Inverse(sourceBody.transform.rotation) * sourceHead.rotation;
            var inverseBindRotation = Quaternion.Inverse(sourceRotation);
            instance.transform.localRotation = inverseBindRotation;
            instance.transform.localPosition = inverseBindRotation * -sourcePosition;
            Debug.Log($"WARDROBE_SOURCE_HEAD owner={option.Owner} position={sourcePosition.ToString("F6")}");
        }
        else
        {
            instance.transform.localPosition = Vector3.zero;
            instance.transform.localRotation = Quaternion.identity;
        }
        instance.transform.localScale = Vector3.one;
        SetLayerRecursively(instance.transform, 0);
        if (string.Equals(category, "Face", StringComparison.OrdinalIgnoreCase) && option.Mesh != null)
        {
            foreach (var renderer in instance.GetComponentsInChildren<MeshRenderer>(true))
            {
                var filter = renderer.GetComponent<MeshFilter>() ?? renderer.gameObject.AddComponent<MeshFilter>();
                filter.sharedMesh = option.Mesh;
                renderer.enabled = true;
            }
        }
        ApplyWardrobeMaterials(instance, category);
        foreach (var renderer in instance.GetComponentsInChildren<SkinnedMeshRenderer>(true))
            renderer.updateWhenOffscreen = true;
        return instance;
    }

    private void ApplyWardrobeMaterials(GameObject instance, string category)
    {
        var reference = CharacterMaterials.Where(material => material != null)
            .ToDictionary(material => material.name.Replace(" 2_fixed", string.Empty), material => material,
                StringComparer.OrdinalIgnoreCase);
        foreach (var renderer in instance.GetComponentsInChildren<Renderer>(true))
        {
            var materials = renderer.sharedMaterials;
            for (var i = 0; i < materials.Length; i++)
            {
                var source = materials[i];
                if (source == null) continue;
                var material = new Material(source) { name = source.name + "_WardrobeRuntime" };
                RuntimeAssetOwner.For(instance).Own(material);
                var key = source.name.Replace(" (Instance)", string.Empty);
                Material template;
                reference.TryGetValue(key, out template);
                if (template == null && string.Equals(category, "Hair", StringComparison.OrdinalIgnoreCase))
                    reference.TryGetValue(key.IndexOf("co", StringComparison.OrdinalIgnoreCase) >= 0 ? "m_hirco" : "m_hir", out template);
                else if (template == null && string.Equals(category, "Body", StringComparison.OrdinalIgnoreCase))
                    reference.TryGetValue("m_bdy", out template);
                else if (template == null)
                    reference.TryGetValue("m_fcp", out template);
                if (template != null && (material.shader == null || material.shader.name == "Hidden/InternalErrorShader"))
                    material.shader = template.shader;
                if (recordedActor) recordedActor.Apply(material, instance.name.StartsWith("Wardrobe HSKI ", StringComparison.Ordinal));
                materials[i] = material;
            }
            renderer.sharedMaterials = materials;
        }
    }

    private void RebuildWardrobeBoneBindings()
    {
        campusHairDirty = true;
        soleRenderers=null;
        cameraRenderers=null;
        sparseSoles.Invalidate();
        if(characterAttributes!=null)characterAttributes.InvalidateRenderers();
        wardrobeBoneBindings.Clear();
        AddWardrobeBindings(activeWardrobeBody, wardrobeTargetBones, "Body");
    }

    private static int WrapIndex(int index, int length)
    {
        return length <= 0 ? 0 : (index % length + length) % length;
    }

    private static int FindOption(WardrobeOption[] options, string bundleName)
    {
        for (var i = 0; i < options.Length; i++)
            if (string.Equals(options[i].BundleName, bundleName, StringComparison.OrdinalIgnoreCase)) return i;
        return 0;
    }

    private void AddWardrobeBindings(GameObject instance, Dictionary<string, Transform> targets, string category)
    {
        if (instance == null) return;
        foreach (var source in instance.GetComponentsInChildren<Transform>(true))
        {
            if (source == instance.transform) continue;
            var isBody = string.Equals(category, "Body", StringComparison.OrdinalIgnoreCase);
            var isFaceAnchor = string.Equals(category, "Face", StringComparison.OrdinalIgnoreCase) &&
                               string.Equals(source.name, "Head_Face", StringComparison.OrdinalIgnoreCase);
            var isHairAnchor = string.Equals(category, "Hair", StringComparison.OrdinalIgnoreCase) &&
                               string.Equals(source.name, "Head_Hair", StringComparison.OrdinalIgnoreCase);
            if (isBody && IsNativeSkirtBone(source.name)) continue;
            if ((isBody || isFaceAnchor || isHairAnchor) && targets.TryGetValue(source.name, out var target))
                wardrobeBoneBindings.Add(new BoneBinding { Source = source, Target = target });
        }
    }

    private void SyncWardrobeBones()
    {
        foreach (var binding in wardrobeBoneBindings)
            if (binding.Source != null && binding.Target != null)
                binding.Source.SetPositionAndRotation(binding.Target.position, binding.Target.rotation);
    }

    private void UpdateCamera(bool immediate)
    {
        if (reference0514)
        {
            playerCamera.transform.SetPositionAndRotation(new Vector3(-0.40078932f,1.37180614f,-4.35819054f),
                Quaternion.LookRotation(new Vector3(0.432106733f,-0.081938557f,0.89809233f), Vector3.up));
            playerCamera.fieldOfView = 44.958872f;
            playerCamera.nearClipPlane = 0.1f;
            playerCamera.farClipPlane = 5000f;
            // Preserve the captured vertical FOV, extend the view horizontally for 16:9.
            playerCamera.aspect = 16f / 9f;
            return;
        }
        var bounds = RuntimePerformance.Legacy ? CalculateBounds(visual) : CachedCameraBounds();
        var focus = new Vector3(bounds.center.x, bounds.min.y + bounds.size.y * 0.56f, bounds.center.z);
        var orbit = Quaternion.Euler(pitch, yaw, 0f);
        var direction = orbit * Vector3.back;
        if (RuntimePerformance.Has("-recordedActorCloseup"))
        {
            focus.y = bounds.min.y + bounds.size.y * .88f;
            playerCamera.transform.SetPositionAndRotation(focus + direction * .85f,
                Quaternion.LookRotation(-direction, Vector3.up));
            return;
        }
        var desired = focus + direction * cameraDistance;
        float nearest=float.PositiveInfinity;
        if(RuntimePerformance.Legacy)
        {
            var hits=Physics.SphereCastAll(focus,0.16f,direction,cameraDistance,~0,QueryTriggerInteraction.Ignore)
                .Where(h=>!h.transform.IsChildOf(playerRoot.transform)&&(!CampusAssetBundleLoader.Indoor||h.collider.bounds.size.y>2f)).OrderBy(h=>h.distance).ToArray();
            if(hits.Length>0)nearest=hits[0].distance;
        }
        else
        {
            int count=Physics.SphereCastNonAlloc(focus,0.16f,direction,cameraHitBuffer,cameraDistance,~0,QueryTriggerInteraction.Ignore);
            // Overflow fallback preserves collision correctness in unusually dense scenes.
            var hits=count==cameraHitBuffer.Length ? Physics.SphereCastAll(focus,0.16f,direction,cameraDistance,~0,QueryTriggerInteraction.Ignore) : cameraHitBuffer;
            if(hits!=cameraHitBuffer)count=hits.Length;
            for(int i=0;i<count;i++){var h=hits[i];if(!h.transform.IsChildOf(playerRoot.transform)&&(!CampusAssetBundleLoader.Indoor||h.collider.bounds.size.y>2f))nearest=Mathf.Min(nearest,h.distance);}
        }
        if(!float.IsPositiveInfinity(nearest))desired=focus+direction*Mathf.Max(0.65f,nearest-0.18f);
        playerCamera.transform.position = immediate
            ? desired
            : Vector3.Lerp(playerCamera.transform.position, desired, 1f - Mathf.Exp(-12f * Time.deltaTime));
        playerCamera.transform.LookAt(focus);
    }

    private static Vector2 SelfTestInput(float time)
    {
        if (time < 1f || time >= 11f) return Vector2.zero;
        if (time < 2f) return Vector2.up;
        if (time < 3f) return Vector2.right;
        if (time < 4f) return Vector2.down;
        if (time < 5f) return Vector2.left;
        if (time < 6f) return Vector2.up;
        if (time < 7f) return Vector2.down;
        if (time < 8.5f) return Vector2.up;
        if (time < 9.2f) return Vector2.zero;
        return new Vector2(0.7f, 0.7f).normalized;
    }

    private static string ResolveFourWayName(Vector2 input)
    {
        if (Mathf.Abs(input.y) >= Mathf.Abs(input.x)) return input.y >= 0f ? "Forward" : "Back";
        return input.x >= 0f ? "Right" : "Left";
    }

    private bool ProbeGrounded()
    {
        var contact = characterController.isGrounded;
        if (!contact && verticalVelocity <= 0f)
        {
            var origin = playerRoot.transform.position + Vector3.up * 0.14f;
            contact = Physics.Raycast(origin, Vector3.down, out var hit, 0.34f, ~0, QueryTriggerInteraction.Ignore)
                      && !hit.transform.IsChildOf(playerRoot.transform);
        }
        if (contact) groundedGrace = 0.08f;
        else groundedGrace = Mathf.Max(0f, groundedGrace - Time.deltaTime);
        return contact || (verticalVelocity <= 0f && groundedGrace > 0f);
    }

    private void ApplyCharacterMaterials(GameObject target)
    {
        var replacements = CharacterMaterials.Where(material => material != null)
            .ToDictionary(material => material.name.Split(' ')[0], material => material, StringComparer.OrdinalIgnoreCase);
        if (recordedActor)
        {
            foreach (var key in replacements.Keys.ToArray())
            {
                var copy = new Material(replacements[key]);
                RuntimeAssetOwner.For(target).Own(copy);
                recordedActor.Apply(copy, true);
                replacements[key] = copy;
            }
        }
        foreach (var renderer in target.GetComponentsInChildren<Renderer>(true))
        {
            var materials = renderer.sharedMaterials;
            var changed = false;
            for (var i = 0; i < materials.Length; i++)
            {
                if (materials[i] == null) continue;
                if (!replacements.TryGetValue(materials[i].name.Split(' ')[0], out var replacement)) continue;
                materials[i] = replacement;
                changed = true;
            }
            if (renderer.name.IndexOf("Geo_Body", StringComparison.OrdinalIgnoreCase) >= 0 && replacements.TryGetValue("m_bdy", out var body))
            { materials = new[] { body }; changed = true; }
            else if (materials.Length == 9)
            {
                var keys = new[] { "m_fcp", "m_fce", "m_fce", "m_fcp", "m_ebs", "m_fce", "m_eye", "m_ehl", "m_fce" };
                var mapped = keys.Select(key => replacements.TryGetValue(key, out var value) ? value : null).ToArray();
                if (mapped.All(material => material != null)) { materials = mapped; changed = true; }
            }
            else if (renderer.name.IndexOf("Geo_HairProp", StringComparison.OrdinalIgnoreCase) >= 0 && replacements.ContainsKey("m_hirco") && replacements.ContainsKey("m_hir"))
            { materials = new[] { replacements["m_hirco"], replacements["m_hir"] }; changed = true; }
            else if (renderer.name.IndexOf("Geo_Hair", StringComparison.OrdinalIgnoreCase) >= 0 && replacements.ContainsKey("m_hir"))
            { materials = new[] { replacements["m_hir"] }; changed = true; }
            if (changed) renderer.sharedMaterials = materials;
            renderer.enabled = true;
            renderer.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.On;
            renderer.receiveShadows = true;
        }
    }

    private static void ScaleToHeight(GameObject target, float desiredHeight)
    {
        var renderers = target.GetComponentsInChildren<Renderer>(true);
        if (renderers.Length == 0) return;
        var bounds = renderers[0].bounds;
        foreach (var renderer in renderers.Skip(1)) bounds.Encapsulate(renderer.bounds);
        if (bounds.size.y > 0.01f) target.transform.localScale *= desiredHeight / bounds.size.y;
    }

    private Bounds CachedCameraBounds()
    {
        if (cameraRenderers == null) cameraRenderers = visual.GetComponentsInChildren<Renderer>(true);
        var bounds = new Bounds(visual.transform.position + Vector3.up * 0.8f, Vector3.one * 1.6f);
        bool found = false;
        foreach (var renderer in cameraRenderers)
        {
            if (renderer == null || !renderer.enabled) continue;
            if (!found) { bounds = renderer.bounds; found = true; }
            else bounds.Encapsulate(renderer.bounds);
        }
        return bounds;
    }

    private static Bounds CalculateBounds(GameObject target)
    {
        var renderers = target.GetComponentsInChildren<Renderer>(true)
            .Where(renderer => renderer.enabled).ToArray();
        if (renderers.Length == 0) return new Bounds(target.transform.position + Vector3.up * 0.8f, Vector3.one * 1.6f);
        var bounds = renderers[0].bounds;
        foreach (var renderer in renderers.Skip(1)) bounds.Encapsulate(renderer.bounds);
        return bounds;
    }

    }

    private static void CreateCampusMeshColliders(Transform hiddenComparisonCharacter)
    {
        var scanned = 0;
        var meshColliders = 0;
        var groundProxies = 0;
        var existing = 0;
        var skippedDecoration = 0;
        var unreadableSkipped = 0;
        var failed = 0;
        var walkableBounds = new List<Bounds>();
        var walkableSurfaceHeights = new List<float>();
        var decorationTokens = new[]
        {
            "grass", "flower", "leaf", "leaves", "tree", "bush", "plant",
            "ivy", "nature", "effect", "particle", "shadow", "shielding"
        };

        foreach (var filter in FindObjectsOfType<MeshFilter>(true))
        {
            if (filter == null || filter.sharedMesh == null || !filter.gameObject.activeInHierarchy)
                continue;
            if (hiddenComparisonCharacter != null && filter.transform.IsChildOf(hiddenComparisonCharacter))
                continue;
            var renderer = filter.GetComponent<MeshRenderer>();
            if (renderer == null || !renderer.enabled) continue;
            scanned++;

            if (filter.GetComponent<Collider>() != null)
            {
                existing++;
                continue;
            }

            var path = GetTransformPath(filter.transform).ToLowerInvariant();
            if (decorationTokens.Any(path.Contains))
            {
                skippedDecoration++;
                continue;
            }

            try
            {
                if (filter.sharedMesh.isReadable)
                {
                    if (CampusAssetBundleLoader.Indoor)
                    {
                        var bounds = filter.sharedMesh.bounds;
                        var size = bounds.size;
                        bool floor = filter.sharedMesh.name.StartsWith("Geo_floor", StringComparison.OrdinalIgnoreCase);
                        bool wall = size.y > 1.2f && Mathf.Min(size.x, size.z) < 0.55f;
                        bool furniture = path.Contains("furniture") && size.y > 0.35f && size.y < 2.5f && size.x < 3f && size.z < 3f;
                        if (floor || wall || furniture)
                        {
                            var box = filter.gameObject.AddComponent<BoxCollider>();
                            box.center = bounds.center - (floor ? Vector3.up * 0.08f : Vector3.zero);
                            box.size = new Vector3(Mathf.Max(size.x, 0.10f), floor ? 0.16f : Mathf.Max(size.y, 0.1f), Mathf.Max(size.z, 0.10f));
                            groundProxies++;
                        }
                        continue;
                    }
                    var collider = filter.gameObject.AddComponent<MeshCollider>();
                    collider.sharedMesh = filter.sharedMesh;
                    collider.convex = false;
                    meshColliders++;
                }
                else if (IsGroundSurface(path, filter.sharedMesh.name, renderer.bounds))
                {
                    var surfaceHeight = CreateGroundBoundsProxy(renderer.bounds, groundProxies);
                    walkableBounds.Add(renderer.bounds);
                    walkableSurfaceHeights.Add(surfaceHeight);
                    groundProxies++;
                    Debug.Log($"LOCOMOTION_GROUND_PROXY mesh={filter.sharedMesh.name} bounds={renderer.bounds}");
                }
                else
                {
                    unreadableSkipped++;
                }
            }
            catch (Exception exception)
            {
                failed++;
                Debug.LogWarning($"LOCOMOTION_COLLIDER_COOK_FAILED object={path} error={exception.Message}");
            }
        }

        if (walkableBounds.Count > 0 && !CampusAssetBundleLoader.Indoor)
            CreateCampusSafetyUnderlay(walkableBounds, walkableSurfaceHeights);

        Physics.SyncTransforms();
        Debug.Log($"LOCOMOTION_CAMPUS_COLLIDERS scanned={scanned} mesh={meshColliders} groundProxies={groundProxies} existing={existing} decorationSkipped={skippedDecoration} unreadableSkipped={unreadableSkipped} failed={failed}");
    }

    private static Vector3 FindIndoorSpawn()
    {
        Physics.SyncTransforms();
        // Prefer the central aisle, reject desks and occupied capsule volumes.
        for (int r = 0; r < 7; r++)
            for (int x = -r; x <= r; x++)
                for (int z = -r; z <= r; z++)
                {
                    var origin = new Vector3(x * 0.5f - 1f, 2f, z * 0.5f - 2f);
                    if (!Physics.Raycast(origin, Vector3.down, out var hit, 3f, ~0, QueryTriggerInteraction.Ignore)) continue;
                    if (hit.normal.y < 0.9f || Mathf.Abs(hit.point.y) > 0.35f) continue;
                    var p = hit.point + Vector3.up * 0.06f;
                    if (Physics.CheckCapsule(p + Vector3.up * 0.32f, p + Vector3.up * 1.48f, 0.29f, ~0, QueryTriggerInteraction.Ignore)) continue;
                    Debug.Log($"CLASSROOM_SPAWN_OK position={p} floor={hit.collider.name}");
                    return p;
                }
        throw new InvalidOperationException("No verified classroom floor spawn; refusing unsafe placement.");
    }

    private static bool IsGroundSurface(string path, string meshName, Bounds bounds)
    {
        var name = (path + "/" + meshName).ToLowerInvariant();
        var groundTokens = new[]
        {
            "ground", "floor", "road", "corridor", "pavement", "street",
            "sidewalk", "walkway", "plaza", "square", "court", "bridge", "path"
        };
        return groundTokens.Any(name.Contains)
               && bounds.size.x >= 0.5f && bounds.size.z >= 0.5f;
    }

    private static float CreateGroundBoundsProxy(Bounds bounds, int index)
    {
        const float thickness = 0.24f;
        // Flat ground meshes use their upper face. Corridor meshes often combine
        // rails/walls with the floor, so their walkable plane is near bounds.min.y.
        var surfaceHeight = bounds.size.y <= 1.25f ? bounds.max.y : bounds.min.y + 0.08f;
        var proxy = new GameObject($"Campus Ground Proxy {index:000}");
        proxy.transform.position = new Vector3(bounds.center.x, surfaceHeight - thickness * 0.5f, bounds.center.z);
        var collider = proxy.AddComponent<BoxCollider>();
        collider.size = new Vector3(
            Mathf.Max(0.5f, bounds.size.x + 0.08f),
            thickness,
            Mathf.Max(0.5f, bounds.size.z + 0.08f));
        return surfaceHeight;
    }

    private static void CreateCampusSafetyUnderlay(List<Bounds> regions, List<float> surfaceHeights)
    {
        var coverage = regions[0];
        foreach (var region in regions.Skip(1)) coverage.Encapsulate(region);
        surfaceHeights.Sort();
        var medianSurface = surfaceHeights[surfaceHeights.Count / 2];
        const float underlayDrop = 0.18f;
        const float thickness = 0.20f;
        var underlay = new GameObject("Campus Walkable Safety Underlay");
        underlay.transform.position = new Vector3(
            coverage.center.x, medianSurface - underlayDrop - thickness * 0.5f, coverage.center.z);
        var collider = underlay.AddComponent<BoxCollider>();
        collider.size = new Vector3(coverage.size.x + 1f, thickness, coverage.size.z + 1f);
        Debug.Log($"LOCOMOTION_GROUND_UNDERLAY bounds={coverage} topY={medianSurface - underlayDrop:F3}");
    }

    private static string GetTransformPath(Transform transform)
    {
        var names = new List<string>();
        for (var current = transform; current != null; current = current.parent)
            names.Add(current.name);
        names.Reverse();
        return string.Join("/", names);
    }

    private static void CreateLocomotionCollisionGround()
    {
        // The extracted campus scene contains visual meshes but its original physics
        // scene was not part of the available bundle dump. Add an invisible, explicit
        // collision slab over the validated courtyard route.
        var ground = new GameObject("Locomotion Courtyard Collision");
        ground.transform.position = new Vector3(-15f, 0.04f, 2f);
        var collider = ground.AddComponent<BoxCollider>();
        collider.size = new Vector3(28f, 0.2f, 34f);
        collider.center = Vector3.zero;
    }
