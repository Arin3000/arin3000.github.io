using System;
using UnityEngine;
using Unity.Mathematics;

// Skirt-specific driver, kept separate from the verified hair solver.
public static class SkirtNativeMath {
 [Serializable] public class Setting {
  public string bone,reference;
  public Quaternion initialReferenceRotation;
  public int rotationOrder;
  public Vector3 innerCoefficient,outerCoefficient,limitMin,limitMax;
 }
 public static Quaternion Calculate(Quaternion initial,Quaternion current,Setting s) {
  var relative=current*Quaternion.Inverse(initial);
  var e=ToEulerXYZ(relative)*Mathf.Rad2Deg;
  var q=quaternion.EulerXYZ((float3)e*Mathf.Deg2Rad).value;
  var rotation=new Quaternion(q.x,q.y,q.z,q.w);
  var up=rotation*Vector3.up;
  float bend=2f*Mathf.Atan2(up.z,1f+up.y)*Mathf.Rad2Deg;
  float roll=-2f*Mathf.Atan2(up.x,1f+up.y)*Mathf.Rad2Deg;
  var rebuilt=Quaternion.Euler(e);
  var twist=rebuilt*Quaternion.Inverse(Quaternion.FromToRotation(Vector3.up,up));
  double length=Math.Sqrt((double)twist.x*twist.x+(double)twist.y*twist.y+(double)twist.z*twist.z+(double)twist.w*twist.w);
  float angle=(float)(2*Math.Acos(Math.Max(-1,Math.Min(1,twist.w/length)))*180/Math.PI);
  if(Vector3.Dot(new Vector3(twist.x,twist.y,twist.z),up)<0)angle=-angle;
  var degrees=new Vector3(Range(angle),Range(roll),Range(bend));
  var result=new Vector3(Dual(degrees.x,s.innerCoefficient.x,s.outerCoefficient.x,s.limitMin.x,s.limitMax.x),Dual(degrees.y,s.innerCoefficient.y,s.outerCoefficient.y,s.limitMin.y,s.limitMax.y),Dual(degrees.z,s.innerCoefficient.z,s.outerCoefficient.z,s.limitMin.z,s.limitMax.z))*Mathf.Deg2Rad;
  float p=Mathf.Tan(result.y*.5f),y=Mathf.Tan(-result.z*.5f),scale=2/(p*p+y*y+1);
  return Quaternion.AngleAxis(-result.x*Mathf.Rad2Deg,Vector3.right)*Quaternion.FromToRotation(Vector3.right,new Vector3(scale-1,scale*y,scale*p));
 }
 static float Dual(float d,float i,float o,float min,float max)=>o*d+(i-o)*Mathf.Clamp(d,min,max);
 static float Range(float d)=>d>180?d-360:d< -180?d+360:d;
 static Vector3 ToEulerXYZ(Quaternion q) {
  float x=q.x,y=q.y,z=q.z,w=-q.w;
  return new Vector3(-Mathf.Atan2(2*(w*x-y*z),w*w-x*x-y*y+z*z),-Mathf.Asin(Mathf.Clamp(2*(w*y+x*z),-1,1)),-Mathf.Atan2(2*(w*z-x*y),w*w+x*x-y*y-z*z));
 }
}
