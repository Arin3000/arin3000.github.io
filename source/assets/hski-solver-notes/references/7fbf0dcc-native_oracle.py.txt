                self.current_fromto['dot']=struct.unpack('<f',struct.pack('<I',uc.reg_read(UC_X86_REG_XMM10)&0xffffffff))[0]
            elif address==0x9c4589:
                self.current_fromto['expected']=list(struct.unpack('<4f',uc.mem_read(uc.reg_read(UC_X86_REG_RDI),16)))

        if address in (0x99eda5,0x99edf7,0x99ee44,0x99ee78):
            key={0x99eda5:'headEuler',0x99edf7:'neckEuler',0x99ee44:'headBend',0x99ee78:'neckBend'}[address]
            self.trace[key]=list(struct.unpack('<3f',uc.mem_read(uc.reg_read(UC_X86_REG_RAX),12)))

        elif address==0x98a830:

        elif address in (0x958ab8,0x9596e0,0x959de0,0x958fdc,0x958d58,0x95aee0):
            x=struct.unpack('<d',struct.pack('<Q',uc.reg_read(UC_X86_REG_XMM0)&0xffffffffffffffff))[0]

    def ret(self):
        sp=self.uc.reg_read(UC_X86_REG_RSP)
        dst=struct.unpack('<Q',self.uc.mem_read(sp,8))[0]
        self.uc.reg_write(UC_X86_REG_RSP,sp+8)
        self.uc.reg_write(UC_X86_REG_RIP,dst)
    def xmm(self,index,value):
        self.uc.reg_write(UC_X86_REG_XMM0+index,struct.unpack('<I',struct.pack('<f',value))[0])
    def floats(self,address,values): self.uc.mem_write(address,struct.pack('<'+'f'*len(values),*values))

def reference(c):
    axis=np.array(c['axis']); q=np.array(c['rotation']); p=np.array(c['selfpos']); child=np.array(c['child']); target=np.array(c['target'])
    direction=axis+2*np.cross(q[:3],np.cross(q[:3],axis)+q[3]*axis)
    correction=0.
    if c['kind']==0:
        delta=direction
        if c['pendulum']>1e-5 and c['prange']>1e-5:
            denom=np.linalg.norm(child)*np.linalg.norm(target)
            cosine=abs(np.dot(child,target))/denom if denom else float('nan')
            correction=max(0.,cosine-(1-c['prange']))/c['prange']*c['pendulum']
    else:
        delta=p+direction*c['length']-child
        if c['pendulum']>1e-5 and c['prange']>1e-5:
            correction=(1-min(np.linalg.norm(target-child)*10,c['prange'])/c['prange'])*c['pendulum']
    return delta*(c['stiffness']-correction)
